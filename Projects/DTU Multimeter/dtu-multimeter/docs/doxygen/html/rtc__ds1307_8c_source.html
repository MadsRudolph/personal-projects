27void rtc_init(void)
28{
29 uint8_t sec;
30
31 /* Read register 0 (seconds) */
32 i2c_start(DS1307_ADDR_W);
33 i2c_write(0x00);
34 i2c_start(DS1307_ADDR_R);
35 sec = i2c_read_nack();
36 i2c_stop();

38 /* Clear CH bit (bit 7) to start oscillator */

40 i2c_start(DS1307_ADDR_W);
41 i2c_write(0x00);
42 i2c_write(sec & 0x7F);
43 i2c_stop();
44 }

47void rtc_read(rtc_time_t *t)
48{
49 /* Set register pointer to 0 */
50 i2c_start(DS1307_ADDR_W);
51 i2c_write(0x00);
52
53 /* Sequential read of 7 registers: 6 ACK + 1 NACK */
54 i2c_start(DS1307_ADDR_R);
55 t->sec = bcd_to_dec(i2c_read_ack() & 0x7F); /* mask CH bit */
56 t->min = bcd_to_dec(i2c_read_ack());
57 t->hour = bcd_to_dec(i2c_read_ack() & 0x3F); /* mask 24h mode */
58 t->day = bcd_to_dec(i2c_read_ack());
59 t->date = bcd_to_dec(i2c_read_ack());
60 t->month = bcd_to_dec(i2c_read_ack());
61 t->year = bcd_to_dec(i2c_read_nack());
62 i2c_stop();
63}
