35{
36 /* Red, Green, Yellow LEDs + Buzzer as outputs */
37 LEDR_DDR |= (1 << LEDR_BIT);
38 LEDG_DDR |= (1 << LEDG_BIT);

44 LEDG_PORT &= ~(1 << LEDG_BIT);

94/* ─── FUNC button handler ────────────────────────────────────────── */
95
96static void handle_func_button(void)
97{
98 if (!btn_func_pressed)
99 return;
100 btn_func_pressed = 0;
101
102 /* Cycle: NONE → REL → HOLD → MINMAX → NONE */
103 switch (current_func) {
104 case FUNC_NONE:
105 current_func = FUNC_REL;
106 rel_reference = 0.0f; /* Will be set on next measurement */
107 break;
108 case FUNC_REL:
109 current_func = FUNC_HOLD;
110 break;
111 case FUNC_HOLD:
112 current_func = FUNC_MINMAX;
113 measure_reset_minmax();
114 break;
115 case FUNC_MINMAX:
116 current_func = FUNC_NONE;
117 break;
118 }
119}

156int main(void)
157{
158 /* ─── Disable watchdog (in case bootloader left it on) ──────── */
159 wdt_disable();
160
161 /* ─── Initialize peripherals ────────────────────────────────── */
162 uart_init();
163 uart_puts_P(PSTR("\r\n=== DTU Digital Multimeter v1.0 ===\r\n"));
164 uart_puts_P(PSTR("Initializing...\r\n"));
165
166 spi_init();
167 i2c_init();
168 timer_init();
169 adc_init();
170 mux_init();
171 led_init();
172 ui_init();
173 display_init();
174 rtc_init();
175 scope_init();
176 measure_init();
177 logging_init();
178
179 /* Enable global interrupts */
180 sei();
181
182 /* Splash screen */
183 render_splash();
184 delay_ms(1500);
185
186 uart_puts_P(PSTR("Ready. Press ? for help.\r\n"));
187 led_green(1);
188
189 /* ─── Measurement result buffer ─────────────────────────────── */
190 meas_result_t result;
191 memset(&result, 0, sizeof(result));
192
193 /* ─── Timing for display/logging rates ──────────────────────── */
194 uint32_t last_display_ms = 0;
195 uint32_t last_log_ms = 0;
196
197 static uint8_t hold_frozen = 0;
198
199 /* ═══════════════════════════════════════════════════════════════
200 * Super Loop
201 * ═══════════════════════════════════════════════════════════════ */
202
203 for (;;) {
204 uint32_t now = millis();
205
206 /* ─── Handle buttons ────────────────────────────────────── */
207 ui_poll_buttons();
208 handle_mode_button();
209 handle_func_button();
210 handle_range_button();
211 handle_select_button();
212
213 /* ─── Handle serial commands ────────────────────────────── */
214 ui_process_serial();
215
216 /* ─── Scope mode (separate path) ────────────────────────── */
217 if (current_mode == M_SCOPE) {
218 if (scope_running) {
219 scope_capture();
220 scope_meas_t sm;
221 scope_auto_measure(&sm);
222 }
223
224 if (now - last_display_ms >= 100) {
225 last_display_ms = now;
226 scope_render_oled(display_buffer());
227 render_scope();
228 }
229 continue;
230 }
231
232 /* ─── Normal measurement modes ──────────────────────────── */
233
234 /* Skip measurement if HOLD is active and we already have a value */
235 if (current_func == FUNC_HOLD && hold_frozen) {
236 /* Still update display at normal rate */
237 if (now - last_display_ms >= 200) {
238 last_display_ms = now;
239 render_measurement(&result);
240 }
241 continue;
242 }
243
244 /* Perform measurement */
245 measure_dispatch(&result);
246
247 /* Capture REL reference on first measurement after enabling */
248 if (current_func == FUNC_REL && rel_reference == 0.0f) {
249 rel_reference = result.primary + rel_reference;
250 /* Re-apply to get delta = 0 on first reading */
251 }
252
253 /* HOLD: freeze after first measurement */
254 if (current_func == FUNC_HOLD) {
255 hold_frozen = 1;
256 } else {
257 hold_frozen = 0;
258 }
259
260 /* ─── Continuity buzzer (fast response) ────────────────── */
261 if (current_mode == M_CONTIN) {
262 buzzer(result.primary < CONTIN_THRESHOLD);
263 } else {
264 buzzer(0);
265 }
266
267 /* ─── LED indicators ────────────────────────────────────── */
268 led_red(result.overload);
269 led_green(!result.overload);
270 led_yellow(current_func == FUNC_HOLD || logging_enabled);
271
272 /* ─── Update display (~5 Hz) ────────────────────────────── */
273 if (now - last_display_ms >= 200) {
274 last_display_ms = now;
275 render_measurement(&result);
276 }
277
278 /* ─── Data logging (1 sample/sec) ───────────────────────── */
279 if (logging_enabled && (now - last_log_ms >= 1000)) {
280 last_log_ms = now;
281 logging_store(result.primary, result.unit);
282 }
283 }
284
285 return 0; /* Never reached */
286}
