6 * display.c — SSD1306 128x64 I2C OLED Framebuffer Driver
7 * Target: ATmega2560 bare-metal AVR C (no Arduino)

31/* ─── SSD1306 Init Sequence (PROGMEM) ────────────────────────────────────── */
32static const uint8_t init_cmds[] PROGMEM = {
33 0xAE, /* Display OFF */
34 0xD5, 0x80, /* Set display clock div: default fosc / 1 */
35 0xA8, 0x3F, /* Set multiplex ratio: 64-1 */
36 0xD3, 0x00, /* Set display offset: none */
37 0x40, /* Set start line: 0 */
38 0x8D, 0x14, /* Charge pump: enable (internal VCC) */
39 0x20, 0x00, /* Memory addressing mode: horizontal */
40 0xA1, /* Segment re-map: col 127 = SEG0 */

42 0xDA, 0x12, /* COM pins config: alternative, no L/R remap */
43 0x81, 0xCF, /* Contrast: 207 */
44 0xD9, 0xF1, /* Pre-charge period: phase1=1, phase2=15 */
45 0xDB, 0x40, /* VCOMH deselect level: ~0.89 x VCC */

68/* ─── display_init ───────────────────────────────────────────────────────── */
69void display_init(void)
70{
71 /* Send the entire PROGMEM init sequence */
72 for (uint8_t i = 0; i < sizeof(init_cmds); i++)
73 display_cmd(pgm_read_byte(&init_cmds[i]));
74
75 display_clear();

77}

85/* ─── display_flush ──────────────────────────────────────────────────────── */
86void display_flush(void)
87{
88 /* Set column address range 0..127 */
89 display_cmd(0x21);
90 display_cmd(0);
91 display_cmd(OLED_WIDTH - 1);
92
93 /* Set page address range 0..7 */
94 display_cmd(0x22);
95 display_cmd(0);
96 display_cmd(OLED_PAGES - 1);
97
98 /* Send framebuffer in 16-byte chunks (64 transactions x 16 bytes) */
99 for (uint16_t offset = 0; offset < sizeof(fb); offset += CHUNK_SIZE) {
100 i2c_start(SSD1306_WRITE_ADDR);
101 i2c_write(CTRL_DATA);
102 for (uint8_t j = 0; j < CHUNK_SIZE; j++)
103 i2c_write(fb[offset + j]);
104 i2c_stop();
105 }
106}

108/* ─── display_pixel ──────────────────────────────────────────────────────── */
109void display_pixel(uint8_t x, uint8_t y, uint8_t on)
110{
111 if (x >= OLED_WIDTH || y >= OLED_HEIGHT)
112 return;
113
114 uint16_t idx = x + (uint16_t)(y >> 3) * OLED_WIDTH;
115 uint8_t bit = (1 << (y & 7));
116
117 if (on)
118 fb[idx] |= bit;
119 else
120 fb[idx] &= ~bit;
121}

159/* ─── display_str_large (2x scaled: 10x14 per character) ─────────────────── */
160void display_str_large(uint8_t x, uint8_t y, const char *s)
161{
162 while (*s) {
163 char c = *s++;
164 if (c < 32 || c > 126)
165 c = '?';
166
167 uint16_t base = (uint16_t)(c - 32) * 5;
168
169 for (uint8_t col = 0; col < 5; col++) {
170 uint8_t line = pgm_read_byte(&font5x7[base + col]);
171 for (uint8_t row = 0; row < 7; row++) {
172 if (line & (1 << row)) {
173 /* Draw a 2x2 block for each set pixel */
174 uint8_t px = x + col * 2;
175 uint8_t py = y + row * 2;
176 display_pixel(px, py, 1);
177 display_pixel(px + 1, py, 1);
178 display_pixel(px, py + 1, 1);
179 display_pixel(px + 1, py + 1, 1);
180 }
181 }
182 }
183 x += 12; /* 10px char + 2px gap */
184 }
185}
