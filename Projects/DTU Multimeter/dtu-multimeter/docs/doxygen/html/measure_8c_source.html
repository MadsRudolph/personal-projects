104/* ═══════════════════════════════════════════════════════════════════
105 * Mode 0: DC Voltage
106 * ═══════════════════════════════════════════════════════════════════ */
107
108void meas_dcv(meas_result_t *r)
109{
110 /* Try low range first (direct, 0-5V, better resolution) */
111 uint16_t raw_lo = adc_read_oversample(ADC_CH_VOLT_LO);
112 float v_lo = adc_to_voltage_os(raw_lo);
113
114 if (v_lo < 4.5f) {
115 r->primary = apply_lpf(v_lo);
116 set_unit(r, "V");
117 r->overload = 0;
118 } else {
119 /* High range via voltage divider */
120 uint16_t raw_hi = adc_read_oversample(ADC_CH_VOLT_HI);
121 float v_hi = adc_to_voltage_os(raw_hi) * VDIV_RATIO;
122 r->primary = apply_lpf(v_hi);
123 set_unit(r, "V");
124 r->overload = (v_hi > 54.0f) ? 1 : 0;
125 }
126 r->secondary = 0.0f;
127 r->range = (r->primary < 4.5f) ? 0 : 1;
128}

190/* ═══════════════════════════════════════════════════════════════════
191 * Mode 5: Resistance (8 auto-ranges)
192 * ═══════════════════════════════════════════════════════════════════ */
193
194void meas_resistance(meas_result_t *r)
195{
196 /* Select reference resistor via mux */
197 mux_select(res_range);
198
199 uint16_t raw = adc_read_oversample(ADC_CH_RESIST);
200 float v = adc_to_voltage_os(raw);
201 float ref = pgm_read_float_near(&rref[res_range]);
202
203 /* R_unknown = R_ref * V_adc / (V_REF - V_adc) */
204 float denom = V_REF - v;
205 if (denom < 0.001f) {
206 r->primary = 99999999.0f;
207 r->overload = 1;
208 } else {
209 r->primary = apply_lpf(ref * v / denom);
210 r->overload = 0;
211 }
212
213 /* Auto-range (unless locked) */
214 if (!range_locked && !autorange_settling()) {
215 uint16_t raw12 = adc_read_raw(ADC_CH_RESIST);
216 int8_t delta = autorange_check(raw12);
217 uint8_t new_range = autorange_clamp(res_range, delta, N_RES_RANGES);
218 if (new_range != res_range) {
219 res_range = new_range;
220 autorange_settle_start();
221 }
222 }
223
224 /* Format unit based on magnitude */
225 if (r->primary >= 1e6f) {
226 r->primary /= 1e6f;
227 set_unit(r, "MOhm");
228 r->prefix = 'M';
229 } else if (r->primary >= 1e3f) {
230 r->primary /= 1e3f;
231 set_unit(r, "kOhm");
232 r->prefix = 'k';
233 } else {
234 set_unit(r, "Ohm");
235 r->prefix = ' ';
236 }
237 r->secondary = 0.0f;
238 r->range = res_range;
239}

346/* ═══════════════════════════════════════════════════════════════════
347 * Mode 10: Capacitance (RC Timing)
348 * ═══════════════════════════════════════════════════════════════════ */
349
350void meas_capacitance(meas_result_t *r)
351{
352 float r_charge = pgm_read_float_near(&cap_r[cap_range]);
353
354 /* Step 1: Fully discharge */
355 cap_discharge();
356 delay_ms(50);
357
358 /* Step 2: Start charging */
359 cap_charge();
360
361 /* Step 3: Time until ADC reads 63.2% of max
362 * 63.2% of 4095 = 2588 (12-bit) */
363 uint16_t threshold = 2588;
364 uint32_t t_start = micros();
365 uint32_t timeout = t_start + 5000000UL; /* 5 second timeout */
366
367 while (1) {
368 uint16_t raw = adc_read_raw(ADC_CH_RESIST);
369 if (raw >= threshold)
370 break;
371 if (micros() > timeout) {
372 /* Timeout — capacitor too large for this range */
373 cap_float();
374 r->primary = 0.0f;
375 r->overload = 1;
376 set_unit(r, "F");
377 r->range = cap_range;
378 return;
379 }
380 }
381
382 uint32_t t_end = micros();
383 cap_float();
384
385 /* tau = time in seconds */
386 float tau = (float)(t_end - t_start) / 1e6f;
387
388 /* C = tau / R */
389 float cap = tau / r_charge;
390
391 /* Auto-range based on timing */
392 if (!range_locked) {
393 if (tau < 0.001f && cap_range > 0) {
394 cap_range--;
395 autorange_settle_start();
396 } else if (tau > 2.0f && cap_range < N_CAP_RANGES - 1) {
397 cap_range++;
398 autorange_settle_start();
399 }
400 }
401
402 /* Format unit */
403 if (cap >= 1e-3f) {
404 r->primary = cap * 1e3f;
405 set_unit(r, "mF");
406 r->prefix = 'm';
407 } else if (cap >= 1e-6f) {
408 r->primary = cap * 1e6f;
409 set_unit(r, "uF");
410 r->prefix = 'u';
411 } else {
412 r->primary = cap * 1e9f;
413 set_unit(r, "nF");
414 r->prefix = 'n';
415 }
416
417 r->primary = apply_lpf(r->primary);
418 r->secondary = 0.0f;
419 r->overload = 0;
420 r->range = cap_range;
421}

423/* ═══════════════════════════════════════════════════════════════════
424 * Mode 11: Inductance (RL Step-Response Timing)
425 * ═══════════════════════════════════════════════════════════════════ */
426
427void meas_inductance(meas_result_t *r)
428{
429 float r_series = pgm_read_float_near(&ind_r[ind_range]);
430 uint8_t mux_ch = pgm_read_byte(&ind_mux[ind_range]);
431
432 /* Select series resistor via mux */
433 mux_select(mux_ch);
434
435 /* Discharge any stored energy — short inductor through R_series */
436 cap_discharge();
437 delay_ms(50);
438
439 /* Read baseline (zero-current) ADC value */
440 uint16_t baseline = adc_read_raw(ADC_CH_RESIST);
441
442 /* Apply step voltage, wait for steady state to measure final value */
443 cap_charge();
444 delay_ms(100);
445 uint16_t final_raw = adc_read_raw(ADC_CH_RESIST);
446 cap_float();
447
448 /* Check for open circuit / no inductor */
449 uint16_t delta = (final_raw > baseline) ? (final_raw - baseline) : 0;
450 if (delta < 20) {
451 mux_disable();
452 r->primary = 0.0f;
453 r->overload = 1;
454 set_unit(r, "H");
455 r->range = ind_range;
456 return;
457 }
458
459 /* Compute 63.2% threshold */
460 uint16_t threshold = baseline + (uint16_t)((float)delta * 0.632f);
461
462 /* Now do the timed measurement: discharge, then step and time */
463 cap_discharge();
464 delay_ms(50);
465
466 cap_charge();
467 uint32_t t_start = micros();
468 uint32_t timeout = t_start + 5000000UL; /* 5 second timeout */
469
470 while (1) {
471 uint16_t raw = adc_read_raw(ADC_CH_RESIST);
472 if (raw >= threshold)
473 break;
474 if (micros() > timeout) {
475 cap_float();
476 mux_disable();
477 r->primary = 0.0f;
478 r->overload = 1;
479 set_unit(r, "H");
480 r->range = ind_range;
481 return;
482 }
483 }
484
485 uint32_t t_end = micros();
486 cap_float();
487 mux_disable();
488
489 /* tau = L / R → L = tau * R */
490 float tau = (float)(t_end - t_start) / 1e6f;
491 float inductance = tau * r_series;
492
493 /* Auto-range based on tau.
494 * Note: for inductance, SMALLER R gives LONGER tau (tau = L/R),
495 * so we range towards smaller R when tau is too short. */
496 if (!range_locked) {
497 if (tau < 0.0001f && ind_range < N_IND_RANGES - 1) {
498 ind_range++; /* smaller R → longer tau */
499 autorange_settle_start();
500 } else if (tau > 2.0f && ind_range > 0) {
501 ind_range--; /* larger R → shorter tau */
502 autorange_settle_start();
503 }
504 }
505
506 /* Format unit */
507 if (inductance >= 1.0f) {
508 r->primary = inductance;
509 set_unit(r, "H");
510 r->prefix = ' ';
511 } else if (inductance >= 1e-3f) {
512 r->primary = inductance * 1e3f;
513 set_unit(r, "mH");
514 r->prefix = 'm';
515 } else {
516 r->primary = inductance * 1e6f;
517 set_unit(r, "uH");
518 r->prefix = 'u';
519 }
520
521 r->primary = apply_lpf(r->primary);
522 r->secondary = 0.0f;
523 r->overload = 0;
524 r->range = ind_range;
525}

527/* ═══════════════════════════════════════════════════════════════════
528 * Mode 12: DC Current (6 auto-ranges)
529 * ═══════════════════════════════════════════════════════════════════ */
530
531void meas_dc_current(meas_result_t *r)
532{
533 cur_select(cur_range);
534
535 uint16_t raw = adc_read_oversample(ADC_CH_CURRENT);
536 float v = adc_to_voltage_os(raw);
537 float shunt = pgm_read_float_near(&ishunt[cur_range]);
538
539 /* Gain is 10x for lowest shunt (0.1 Ohm), 1x otherwise */
540 float gain = (cur_range == N_CUR_RANGES - 1) ? CUR_GAIN_HI : CUR_GAIN_LO;
541
542 /* I = V_adc / (R_shunt * Gain) */
543 float current = v / (shunt * gain);
544
545 /* Auto-range */
546 if (!range_locked && !autorange_settling()) {
547 uint16_t raw12 = adc_read_raw(ADC_CH_CURRENT);
548 int8_t delta = autorange_check(raw12);
549 uint8_t new_range = autorange_clamp(cur_range, delta, N_CUR_RANGES);
550 if (new_range != cur_range) {
551 cur_range = new_range;
552 autorange_settle_start();
553 }
554 }
555
556 /* Format unit */
557 if (current >= 1.0f) {
558 r->primary = apply_lpf(current);
559 set_unit(r, "A");
560 r->prefix = ' ';
561 } else if (current >= 0.001f) {
562 r->primary = apply_lpf(current * 1000.0f);
563 set_unit(r, "mA");
564 r->prefix = 'm';
565 } else {
566 r->primary = apply_lpf(current * 1e6f);
567 set_unit(r, "uA");
568 r->prefix = 'u';
569 }
570
571 r->secondary = 0.0f;
572 r->overload = (v > 4.9f) ? 1 : 0;
573 r->range = cur_range;
574}

656/* ═══════════════════════════════════════════════════════════════════
657 * Mode 15: Duty Cycle
658 * ═══════════════════════════════════════════════════════════════════ */
659
660extern volatile uint32_t freq_high_us;
661
662void meas_duty_cycle(meas_result_t *r)
663{
664 if (freq_ready && freq_period_us > 0) {
665 freq_ready = 0;
666 float duty = (float)freq_high_us / (float)freq_period_us * 100.0f;

742void meas_dbm(meas_result_t *r)

812void measure_dispatch(meas_result_t *result)
813{
814 mode_t mode = current_mode;
815 if (mode >= M_SCOPE)
816 return; /* Scope handled separately */
817
818 meas_fn fn;
819 memcpy_P(&fn, &dispatch_table[mode], sizeof(meas_fn));
820 fn(result);
821
822 /* Apply secondary functions */
823 measure_apply_secondary(result);
824}

826/* ═══════════════════════════════════════════════════════════════════
827 * Secondary Function Processing (REL, HOLD, MIN/MAX)
828 * ═══════════════════════════════════════════════════════════════════ */
829
830void measure_apply_secondary(meas_result_t *r)
831{
832 /* REL mode: subtract stored reference */
833 if (current_func == FUNC_REL) {
834 r->primary -= rel_reference;
835 }
836
837 /* Min/Max tracking */
838 if (current_func == FUNC_MINMAX) {
839 if (r->primary < min_val) min_val = r->primary;
840 if (r->primary > max_val) max_val = r->primary;
841 avg_sum += (double)r->primary;
842 avg_count++;
843 }
844
845 /* HOLD mode: don't update primary (handled in display layer) */
846}

848void measure_reset_minmax(void)
849{
850 min_val = 1e30f;
851 max_val = -1e30f;
852 avg_sum = 0.0;
853 avg_count = 0;
854}
