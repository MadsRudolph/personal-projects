5/* ═══════════════════════════════════════════════════════════════════
6 * trms.c — True-RMS Computation Engine
7 * Target: ATmega2560, bare-metal (no Arduino framework)
8 *
9 * Samples an MCP3208 12-bit ADC channel at 20 kHz (50 us intervals)
10 * and computes True-RMS, DC, AC, and peak values.
11 *
12 * Uses a double (64-bit) accumulator for the sum-of-squares to
13 * avoid overflow: 400 samples x 4095^2 ~ 6.7e9, which exceeds
14 * the 24-bit mantissa of a float but fits comfortably in a double.
15 * ═══════════════════════════════════════════════════════════════════ */
16

23/* Scale factor: convert 12-bit raw value to voltage */
24#define RAW_TO_V (V_REF / 4095.0f)
25
26/* ─── trms_measure ────────────────────────────────────────────────
27 * 1. Acquire TRMS_SAMPLES readings at TRMS_INTERVAL_US spacing
28 * 2. Track min/max during acquisition
29 * 3. Compute RMS, DC, AC, convert peaks to voltage
30 * 4. Fill result struct
31 * ──────────────────────────────────────────────────────────────── */
32void trms_measure(uint8_t adc_channel, trms_result_t *result)
33{
34 static uint16_t buf[TRMS_SAMPLES]; /* static — 800 B off the stack */
35
36 uint16_t raw_min = 4095;
37 uint16_t raw_max = 0;
38
39 /* ── Acquisition loop ─────────────────────────────────────── */
40 uint32_t next_time = micros();
41
42 for (uint16_t i = 0; i < TRMS_SAMPLES; i++) {
43 /* Busy-wait until the next sample instant */
44 while (micros() < next_time)
45 ;
46
47 uint16_t raw = adc_read_raw(adc_channel);
48 buf[i] = raw;
49
50 if (raw < raw_min) raw_min = raw;
51 if (raw > raw_max) raw_max = raw;
52
53 next_time += TRMS_INTERVAL_US;
54 }
55
56 /* ── Compute DC (mean) ────────────────────────────────────── */
57 result->dc = trms_compute_dc(buf, TRMS_SAMPLES);
58
59 /* ── Compute True-RMS ─────────────────────────────────────── */
60 result->rms = trms_compute_rms(buf, TRMS_SAMPLES);
61
62 /* ── AC component ─────────────────────────────────────────── */
63 float rms2 = result->rms * result->rms;
64 float dc2 = result->dc * result->dc;
65
66 if (rms2 > dc2)
67 result->ac = sqrtf(rms2 - dc2);
68 else
69 result->ac = 0.0f; /* rounding guard */
70
71 /* ── Peaks → voltage ──────────────────────────────────────── */
72 result->peak_pos = (float)raw_max * RAW_TO_V;
73 result->peak_neg = (float)raw_min * RAW_TO_V;
74}
75
76/* ─── trms_compute_rms ────────────────────────────────────────────
77 * RMS = sqrt( (1/N) * sum(raw_i^2) ) * V_REF / 4095
78 *
79 * A double accumulator is used because the worst-case sum
80 * (400 * 4095^2 ≈ 6.7 x 10^9) exceeds 32-bit float precision.
81 * ──────────────────────────────────────────────────────────────── */
82float trms_compute_rms(const uint16_t *samples, uint16_t count)
83{
84 double sum_sq = 0.0;
85
86 for (uint16_t i = 0; i < count; i++) {
87 double s = (double)samples[i];
88 sum_sq += s * s;
89 }
90
91 float rms_raw = sqrtf((float)(sum_sq / (double)count));
92 return rms_raw * RAW_TO_V;
93}
94
95/* ─── trms_compute_dc ─────────────────────────────────────────────
96 * DC = (1/N) * sum(raw_i) * V_REF / 4095
97 *
98 * A uint32_t accumulator suffices: 400 * 4095 = 1 638 000,
99 * well within 32-bit range.
100 * ──────────────────────────────────────────────────────────────── */
101float trms_compute_dc(const uint16_t *samples, uint16_t count)
102{
103 uint32_t sum = 0;
104
105 for (uint16_t i = 0; i < count; i++)
106 sum += samples[i];
107
108 float mean_raw = (float)sum / (float)count;
109 return mean_raw * RAW_TO_V;
110}
