29/* ─── Single 12-bit read ──────────────────────────────────────────── */
30
31uint16_t adc_read_raw(uint8_t channel)
32{
33 uint8_t byte1, byte2, byte3;
34
35 ADC_CS_LOW();
36 _delay_us(1); /* t_SUCS settling time */
37
38 /* MCP3208 SPI frame (single-ended):
39 * byte1: 0 0 0 0 0 1 SGL D2
40 * byte2: D1 D0 x x x x x x
41 * byte3: clocks out remaining result bits
42 *
43 * SGL=1 for single-ended → leading nibble = 0x06
44 * D2 is the MSB of the 3-bit channel number.
45 */
46 byte1 = spi_transfer(0x06 | (channel >> 2));
47 byte2 = spi_transfer((channel & 0x03) << 6);
48 byte3 = spi_transfer(0x00);
49
50 ADC_CS_HIGH();
51
52 /* byte2 lower 4 bits = B11..B8, byte3 = B7..B0 */
53 return (uint16_t)((byte2 & 0x0F) << 8) | byte3;
54}
