26 DDRB |= (1 << PB0) | (1 << PB1) | (1 << PB2);
27 DDRB &= ~(1 << PB3);

29 /* Drive SS high (deselect) */

32 /*
33 * SPCR: SPE = 1 (enable SPI)
34 * MSTR = 1 (master mode)
35 * CPOL = 0 (Mode 0,0)
36 * CPHA = 0
37 * SPR0 = 1 (base prescaler /8 with SPI2X)
38 * SPR1 = 0
39 *
40 * Prescaler: SPR1:0 = 01 gives /16 base.
41 * SPI2X = 1 halves to /8 → 16 MHz / 8 = 2 MHz.
42 */
43 SPCR = (1 << SPE) | (1 << MSTR) | (1 << SPR0);
