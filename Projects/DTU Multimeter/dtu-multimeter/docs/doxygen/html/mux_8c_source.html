9/* ===================================================================
10 * Multiplexer & Switch Control Driver
11 *
12 * 74HC4067 16-ch analog mux — PA0-PA4
13 * CD4053 triple 2:1 switch — PA5-PA7, PC7
14 * NE555 current source enable — PC4
15 * Capacitance charge/discharge — PC5, PC6
16 *
17 * All functions use read-modify-write bit masking so that only the
18 * relevant pins are touched; other bits on the same port are
19 * preserved.
20 * =================================================================== */

22/* Bit masks for convenience */
23#define MUX_SEL_MASK ((1 << MUX_S0) | (1 << MUX_S1) | \
24 (1 << MUX_S2) | (1 << MUX_S3))

26#define CUR_ABC_MASK ((1 << CUR_A) | (1 << CUR_B) | (1 << CUR_C))

28#define MUX_SETTLE_US 50

31/* ─── mux_init ───────────────────────────────────────────────────── */
32void mux_init(void)
33{
34 /* 74HC4067: S0-S3 and EN as outputs */
35 MUX_DDR |= MUX_SEL_MASK | (1 << MUX_EN);
36
37 /* Disable mux: EN = HIGH (active low) */
38 MUX_PORT |= (1 << MUX_EN);
39
40 /* Clear channel select bits */
41 MUX_PORT &= ~MUX_SEL_MASK;
42
43 /* CD4053: A, B, C as outputs */
44 CUR_ABC_DDR |= CUR_ABC_MASK;
45
46 /* Clear A/B/C */
47 CUR_ABC_PORT &= ~CUR_ABC_MASK;
48
49 /* CD4053 INH as output, set HIGH (inhibit) */
50 CUR_INH_DDR |= (1 << CUR_INH_BIT);
51 CUR_INH_PORT |= (1 << CUR_INH_BIT);
52
53 /* NE555 enable pin as output, LOW (off) */
54 ISRC_DDR |= (1 << ISRC_BIT);
55 ISRC_PORT &= ~(1 << ISRC_BIT);
56
57 /* Capacitance pins: input (Hi-Z), no pull-ups */
58 CAP_CHG_DDR &= ~(1 << CAP_CHG_BIT);
59 CAP_CHG_PORT &= ~(1 << CAP_CHG_BIT);
60 CAP_DIS_DDR &= ~(1 << CAP_DIS_BIT);
61 CAP_DIS_PORT &= ~(1 << CAP_DIS_BIT);
62}

68/* ─── mux_select ─────────────────────────────────────────────────── */
69void mux_select(uint8_t channel)
70{
71 uint8_t tmp;
72
73 /* Clamp to valid range */
74 channel &= 0x0F;
75
76 /* Read current port state */
77 tmp = MUX_PORT;
78
79 /* Clear S0-S3 bits */
80 tmp &= ~MUX_SEL_MASK;
81
82 /* Set new channel (S0-S3 are at PA0-PA3, so channel maps 1:1) */
83 tmp |= channel;
84
85 /* Enable mux: clear EN bit (active low) */
86 tmp &= ~(1 << MUX_EN);
87
88 MUX_PORT = tmp;
89
90 _delay_us(MUX_SETTLE_US);
91}

100/* ═══════════════════════════════════════════════════════════════════
101 * CD4053 — triple 2:1 analog switch (current shunt selection)
102 *
103 * range 0-5 maps to a 3-bit binary value on pins A(PA5), B(PA6),
104 * C(PA7). The three switch sections route different shunt resistors
105 * into the measurement path.
106 * ═══════════════════════════════════════════════════════════════════ */
107
108/* ─── cur_select ─────────────────────────────────────────────────── */
109void cur_select(uint8_t range)
110{
111 uint8_t tmp;
112
113 /* Clamp to 0-5 (6 valid current ranges) */
114 if (range > 5)
115 range = 5;
116
117 tmp = CUR_ABC_PORT;
118
119 /* Clear A/B/C bits (PA5-PA7) */
120 tmp &= ~CUR_ABC_MASK;
121
122 /* Map range to A/B/C: shift the 3-bit value up to PA5 position */
123 tmp |= ((uint8_t)range << CUR_A);
124
125 CUR_ABC_PORT = tmp;
126
127 /* Uninhibit: drive INH low */
128 CUR_INH_PORT &= ~(1 << CUR_INH_BIT);
129
130 _delay_us(CUR_SETTLE_US);
131}

158 *
159 * charge: CHG pin HIGH (output), DIS pin Hi-Z (input)
160 * discharge: DIS pin LOW (output), CHG pin LOW (output)

162 * ═══════════════════════════════════════════════════════════════════ */

164/* ─── cap_charge ─────────────────────────────────────────────────── */
165void cap_charge(void)
166{
167 /* Discharge pin to Hi-Z: clear DDR bit (input), no pull-up */
168 CAP_DIS_DDR &= ~(1 << CAP_DIS_BIT);
169 CAP_DIS_PORT &= ~(1 << CAP_DIS_BIT);
170
171 /* Charge pin as output, drive HIGH */
172 CAP_CHG_DDR |= (1 << CAP_CHG_BIT);
173 CAP_CHG_PORT |= (1 << CAP_CHG_BIT);
174}

176/* ─── cap_discharge ──────────────────────────────────────────────── */
177void cap_discharge(void)
178{
179 /* Charge pin LOW (output) */
180 CAP_CHG_DDR |= (1 << CAP_CHG_BIT);
181 CAP_CHG_PORT &= ~(1 << CAP_CHG_BIT);
182
183 /* Discharge pin as output, drive LOW */
184 CAP_DIS_DDR |= (1 << CAP_DIS_BIT);
185 CAP_DIS_PORT &= ~(1 << CAP_DIS_BIT);
186}

188/* ─── cap_float ──────────────────────────────────────────────────── */
189void cap_float(void)
190{
191 /* Both pins to Hi-Z: input mode, no pull-ups */
192 CAP_CHG_DDR &= ~(1 << CAP_CHG_BIT);
193 CAP_CHG_PORT &= ~(1 << CAP_CHG_BIT);
194 CAP_DIS_DDR &= ~(1 << CAP_DIS_BIT);
