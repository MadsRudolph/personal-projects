35/* ═══════════════════════════════════════════════════════════════════
36 * timer_init — configure Timer1 tick and INT4 capture
37 * ═══════════════════════════════════════════════════════════════════ */
38void timer_init(void)
39{
40 /* ── Timer1: CTC, prescaler = 1 ─────────────────────────────── */
41 TCCR1A = 0;
42 TCCR1B = (1 << WGM12) /* CTC — top = OCR1A */
43 | (1 << CS10); /* clk/1 (no prescaler) */
44 OCR1A = 15999; /* 16 000 counts → 1 ms */
45 TIMSK1 = (1 << OCIE1A); /* Compare-match A interrupt */
46
47 /* ── INT4 (PE4): input, pull-up, rising edge ────────────────── */
48 FREQ_DDR &= ~(1 << FREQ_BIT); /* PE4 as input */
49 FREQ_PORT |= (1 << FREQ_BIT); /* internal pull-up */
50
51 EICRB = (EICRB & ~((1 << ISC41) | (1 << ISC40)))
52 | (1 << ISC41) | (1 << ISC40); /* ISC41:40 = 11 → rising */
53
54 waiting_for_falling = 0;
55 freq_ready = 0;
56
57 EIMSK |= (1 << INT4); /* enable INT4 */
58}

80/* ═══════════════════════════════════════════════════════════════════
81 * micros — microseconds since boot
82 *
83 * Timer1 runs at 16 MHz (prescaler = 1), TCNT1 spans 0..15999
84 * within each 1 ms period. TCNT1 / 16 gives the fractional us.
85 * ═══════════════════════════════════════════════════════════════════ */
86uint32_t micros(void)
87{
88 uint32_t ms;
89 uint16_t tcnt;
90
91 ATOMIC_BLOCK(ATOMIC_RESTORESTATE) {
92 ms = millis_counter;
93 tcnt = TCNT1;
94 /*
95 * If the compare-match flag fired but the ISR hasn't run yet,
96 * millis_counter is stale — compensate by adding 1 ms.
97 */
98 if ((TIFR1 & (1 << OCF1A)) && tcnt < 8000) {
99 ms++;
100 }
101 }
102
103 return ms * 1000UL + (tcnt >> 4); /* tcnt / 16 */
104}

116/* ═══════════════════════════════════════════════════════════════════
117 * INT4 ISR — frequency & duty-cycle capture
118 *
119 * Alternates edge sense between rising and falling:
120 * Rising → record timestamp, compute period from previous rising
121 * Falling → record timestamp, compute high-time
122 * ═══════════════════════════════════════════════════════════════════ */
123ISR(INT4_vect)
124{
125 uint32_t now = micros();
126
127 if (!waiting_for_falling) {
128 /* ── Rising edge ─────────────────────────────────────────── */
129 if (last_rising_us != 0) {
130 freq_period_us = now - last_rising_us;
131 freq_ready = 1;
132 }
133 last_rising_us = now;
134
135 /* Switch to falling-edge detect: ISC41:40 = 10 */
136 EICRB = (EICRB & ~(1 << ISC40)) | (1 << ISC41);
137 waiting_for_falling = 1;
138
139 } else {
140 /* ── Falling edge ────────────────────────────────────────── */
141 freq_high_us = now - last_rising_us;
142
143 /* Switch back to rising-edge detect: ISC41:40 = 11 */
144 EICRB |= (1 << ISC41) | (1 << ISC40);
145 waiting_for_falling = 0;
146 }
147}
