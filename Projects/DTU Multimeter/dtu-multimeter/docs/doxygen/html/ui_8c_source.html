38ISR(INT5_vect)
39{
40 uint32_t now = millis();

47/* ─── Initialization ─────────────────────────────────────────────── */
48
49void ui_init(void)
50{
51 /* MODE button: PE5 (D3) as input with pull-up */
52 BTN_MODE_DDR &= ~(1 << BTN_MODE_BIT);
53 BTN_MODE_PORT |= (1 << BTN_MODE_BIT);
54
55 /* FUNC button: PG5 (D4) as input with pull-up */
56 BTN_FUNC_DDR &= ~(1 << BTN_FUNC_BIT);
57 BTN_FUNC_PORT |= (1 << BTN_FUNC_BIT);
58
59 /* RANGE button: PE3 (D5) as input with pull-up */
60 BTN_RANGE_DDR &= ~(1 << BTN_RANGE_BIT);
61 BTN_RANGE_PORT |= (1 << BTN_RANGE_BIT);
62
63 /* SELECT button: PH3 (D6) as input with pull-up */
64 BTN_SEL_DDR &= ~(1 << BTN_SEL_BIT);
65 BTN_SEL_PORT |= (1 << BTN_SEL_BIT);
66
67 /* Configure INT5 (PE5) for falling edge (active-low button) */
68 EICRB |= (1 << ISC51); /* ISC51=1, ISC50=0 → falling edge */
69 EICRB &= ~(1 << ISC50);
70 EIMSK |= (1 << INT5); /* Enable INT5 */
71
72 last_func_ms = 0;
73 last_range_ms = 0;
74 last_sel_ms = 0;
75 last_mode_ms = 0;
76}

78/* ─── Poll buttons (call from main loop) ─────────────────────────── */
79
80void ui_poll_buttons(void)
81{
82 uint32_t now = millis();
83
84 /* FUNC — PG5, active LOW */
85 if (!(BTN_FUNC_PINR & (1 << BTN_FUNC_BIT))) {
86 if (now - last_func_ms >= DEBOUNCE_MS) {
87 last_func_ms = now;
88 btn_func_pressed = 1;
89 }
90 }
91
92 /* RANGE — PE3, active LOW */
93 if (!(BTN_RANGE_PINR & (1 << BTN_RANGE_BIT))) {
94 if (now - last_range_ms >= DEBOUNCE_MS) {
95 last_range_ms = now;
96 btn_range_pressed = 1;
97 }
98 }
99
100 /* SELECT — PH3, active LOW */
101 if (!(BTN_SEL_PINR & (1 << BTN_SEL_BIT))) {
102 if (now - last_sel_ms >= DEBOUNCE_MS) {
103 last_sel_ms = now;
104 btn_sel_pressed = 1;
105 }
106 }
107}

127/* ─── Serial command processing ──────────────────────────────────── */
128
129void ui_process_serial(void)
130{
131 int c = uart_getc_nb();
132 if (c < 0)
133 return;
134
135 switch ((char)c) {
136 case 'R':
137 case 'r':
138 if (current_func == FUNC_REL) {
139 current_func = FUNC_NONE;
140 uart_puts_P(PSTR("REL off\r\n"));
141 } else {
142 current_func = FUNC_REL;
143 /* Store current reading as reference at next measurement */
144 uart_puts_P(PSTR("REL on\r\n"));
145 }
146 break;
147
148 case 'H':
149 case 'h':
150 if (current_func == FUNC_HOLD) {
151 current_func = FUNC_NONE;
152 uart_puts_P(PSTR("HOLD off\r\n"));
153 } else {
154 current_func = FUNC_HOLD;
155 uart_puts_P(PSTR("HOLD on\r\n"));
156 }
157 break;
158
159 case 'M':
160 case 'm':
161 if (current_func == FUNC_MINMAX) {
162 current_func = FUNC_NONE;
163 uart_puts_P(PSTR("MinMax off\r\n"));
164 } else {
165 current_func = FUNC_MINMAX;
166 measure_reset_minmax();
167 uart_puts_P(PSTR("MinMax on\r\n"));
168 }
169 break;
170
171 case 'L':
172 case 'l':
173 lpf_enabled = !lpf_enabled;
174 uart_puts_P(lpf_enabled ? PSTR("LPF on\r\n") : PSTR("LPF off\r\n"));
175 break;
176
177 case 'G':
178 case 'g':
179 logging_enabled = !logging_enabled;
180 uart_puts_P(logging_enabled ? PSTR("LOG on\r\n") : PSTR("LOG off\r\n"));
181 break;
182
183 case 'K':
184 case 'k':
185 range_locked = !range_locked;
186 uart_puts_P(range_locked ? PSTR("Range LOCKED\r\n") : PSTR("Range AUTO\r\n"));
187 break;
188
189 case '+':
190 /* Range up handled by main loop */
191 btn_range_pressed = 1;
192 break;
193
194 case '-':
195 /* Range down — flag for main loop */
196 /* We reuse btn_sel_pressed as range-down trigger from serial */
197 btn_sel_pressed = 1;
198 break;
199
200 case 'Z':
201 case 'z':
202 dbm_impedance_idx = (dbm_impedance_idx + 1) % 3;
203 uart_puts_P(PSTR("dBm Z: "));
204 if (dbm_impedance_idx == 0) uart_puts_P(PSTR("50\r\n"));
205 else if (dbm_impedance_idx == 1) uart_puts_P(PSTR("75\r\n"));
206 else uart_puts_P(PSTR("600\r\n"));
207 break;
208
209 case 'P':
210 case 'p':
211 uart_puts_P(PSTR("Min: "));
212 uart_print_float(min_val, 4);
213 uart_puts_P(PSTR(" Max: "));
214 uart_print_float(max_val, 4);
215 uart_puts_P(PSTR(" Avg: "));
216 if (avg_count > 0)
217 uart_print_float((float)(avg_sum / (double)avg_count), 4);
218 else
219 uart_puts_P(PSTR("N/A"));
220 uart_puts_P(PSTR("\r\n"));
221 break;
222
223 case 'E':
224 case 'e':
225 /* EEPROM dump handled by logging module — flag for main loop */
226 uart_puts_P(PSTR("EEPROM dump:\r\n"));
227 /* logging_dump_eeprom() called from main */
228 break;
229
230 case '?':
231 uart_puts_P(help_text);
232 break;
233
234 default:
235 break;
236 }
