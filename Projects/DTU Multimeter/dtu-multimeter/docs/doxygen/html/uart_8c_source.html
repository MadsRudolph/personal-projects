47/* ═══════════════════════════════════════════════════════════════════
48 * uart_init — Configure UART0: 115200-8N1, U2X0, TX/RX + interrupts
49 * ═══════════════════════════════════════════════════════════════════ */
50void uart_init(void)
51{
52 /* Set baud rate */
53 UBRR0H = (uint8_t)(UBRR_VAL >> 8);
54 UBRR0L = (uint8_t)(UBRR_VAL);
55
56 /* Enable double-speed for lower baud error at 115200 */
57 UCSR0A = (1 << U2X0);
58
59 /* Enable TX, RX, and RX-complete interrupt */
60 UCSR0B = (1 << TXEN0) | (1 << RXEN0) | (1 << RXCIE0);
61
62 /* Frame: 8 data, 1 stop, no parity */
63 UCSR0C = (1 << UCSZ01) | (1 << UCSZ00);
64
65 /* Clear buffer indices */
66 tx_head = 0;
67 tx_tail = 0;
68 rx_head = 0;
69 rx_tail = 0;
70
71 /* Redirect stdout so printf_P works */
72 stdout = &uart_stdout;
73}

75/* ═══════════════════════════════════════════════════════════════════
76 * TX ISR — USART0 Data Register Empty
77 * Sends next byte from TX ring buffer; disables UDRIE when empty.
78 * ═══════════════════════════════════════════════════════════════════ */
79ISR(USART0_UDRE_vect)
80{
81 if (tx_head == tx_tail) {
82 /* Buffer empty — disable Data Register Empty interrupt */
83 UCSR0B &= ~(1 << UDRIE0);
84 return;
85 }
86 UDR0 = tx_buf[tx_tail];
87 tx_tail = (tx_tail + 1) & TX_MASK;
88}
89
90/* ═══════════════════════════════════════════════════════════════════
91 * RX ISR — USART0 Receive Complete
92 * Stores incoming byte into RX ring buffer (drops if full).
93 * ═══════════════════════════════════════════════════════════════════ */
94ISR(USART0_RX_vect)
95{
96 uint8_t data = UDR0;
97 uint8_t next = (rx_head + 1) & RX_MASK;
98
99 if (next != rx_tail) { /* room in buffer? */
100 rx_buf[rx_head] = data;
101 rx_head = next;
102 }
103 /* else: overflow — byte dropped silently */
104}

106/* ═══════════════════════════════════════════════════════════════════
107 * uart_putc — Queue one char into TX buffer (blocks if full)
108 * ═══════════════════════════════════════════════════════════════════ */
109void uart_putc(char c)
110{
111 uint8_t next = (tx_head + 1) & TX_MASK;
112
113 /* Block until space available */
114 while (next == tx_tail)
115 ;
116
117 tx_buf[tx_head] = (uint8_t)c;
118 tx_head = next;
119
120 /* Enable UDRE interrupt to start transmission */
121 UCSR0B |= (1 << UDRIE0);
122}

187/* ═══════════════════════════════════════════════════════════════════
188 * uart_print_float — Print a float with N decimal places
189 * Manual integer+fraction formatting (no dtostrf dependency).
190 * ═══════════════════════════════════════════════════════════════════ */
191void uart_print_float(float val, uint8_t decimals)
192{
193 char buf[16];
194
195 /* Handle negative values */
196 if (val < 0.0f) {
197 uart_putc('-');
198 val = -val;
199 }
200
201 /* Round to requested decimal places */
202 float rounding = 0.5f;
203 for (uint8_t i = 0; i < decimals; i++)
204 rounding /= 10.0f;
205 val += rounding;
206
207 /* Integer part */
208 uint32_t int_part = (uint32_t)val;
209 float remainder = val - (float)int_part;
210
211 /* Convert integer part to string (reverse fill) */
212 uint8_t pos = 0;
213 if (int_part == 0) {
214 buf[pos++] = '0';
215 } else {
216 char tmp[11];
217 uint8_t ti = 0;
218 while (int_part > 0) {
219 tmp[ti++] = '0' + (int_part % 10);
220 int_part /= 10;
221 }
222 /* Reverse into buf */
223 while (ti > 0)
224 buf[pos++] = tmp[--ti];
225 }
226
227 /* Decimal point and fractional digits */
228 if (decimals > 0) {
229 buf[pos++] = '.';
230 for (uint8_t i = 0; i < decimals; i++) {
231 remainder *= 10.0f;
232 uint8_t digit = (uint8_t)remainder;
233 buf[pos++] = '0' + digit;
234 remainder -= digit;
235 }
236 }
237
238 buf[pos] = '\0';
239 uart_puts(buf);
240}
