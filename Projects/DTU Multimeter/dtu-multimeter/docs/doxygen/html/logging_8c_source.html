95/* ─── logging_init ───────────────────────────────────────────────── */
96void logging_init(void)
97{
98 log_index = eeprom_read_byte(EE_ADDR_INDEX);
99 log_count = eeprom_read_byte(EE_ADDR_COUNT);
100
101 /* Sanity check — virgin / corrupted EEPROM reads 0xFF */
102 if (log_index >= LOG_EEPROM_MAX) {
103 log_index = 0;
104 log_count = 0;
105 eeprom_update_byte(EE_ADDR_INDEX, 0);
106 eeprom_update_byte(EE_ADDR_COUNT, 0);
107 }
108 if (log_count > LOG_EEPROM_MAX) {
109 log_count = 0;
110 eeprom_update_byte(EE_ADDR_COUNT, 0);
111 }
112}

114/* ─── logging_store ──────────────────────────────────────────────── */
115void logging_store(float value, const char *unit)
116{
117 rtc_time_t t;
118 uint8_t *base;
119
120 /* 1. Read current time from DS1307 */
121 rtc_read(&t);
122
123 /* 2. Write entry to EEPROM at current index */
124 base = ee_entry_addr(log_index);
125
126 ee_write_float(base, value); /* bytes 0-3: value */
127 eeprom_update_byte(base + 4, t.hour); /* byte 4: hour */
128 eeprom_update_byte(base + 5, t.min); /* byte 5: minute */
129 eeprom_update_byte(base + 6, t.sec); /* byte 6: second */
130 eeprom_update_byte(base + 7, t.date); /* byte 7: date */
131 eeprom_update_byte(base + 8, t.month); /* byte 8: month */
132 eeprom_update_byte(base + 9, t.year); /* byte 9: year */
133
134 /* 3. Advance index (wrap), cap count */
135 log_index++;
136 if (log_index >= LOG_EEPROM_MAX)
137 log_index = 0;
138
139 if (log_count < LOG_EEPROM_MAX)
140 log_count++;
141
142 /* 4. Persist index and count */
143 eeprom_update_byte(EE_ADDR_INDEX, log_index);
144 eeprom_update_byte(EE_ADDR_COUNT, log_count);
145
146 /* 5. Print CSV line to UART */
147 /* unit is expected to be a PROGMEM pointer from the caller */
148 print_entry(value, t.hour, t.min, t.sec,
149 t.date, t.month, t.year, unit);
150}

152/* ─── logging_dump_eeprom ────────────────────────────────────────── */
153void logging_dump_eeprom(void)
154{
155 uint8_t i, pos;
156 uint8_t *base;
157 float val;
158 uint8_t hour, min, sec, date, month, year;
159
160 if (log_count == 0) {
161 uart_puts_P(msg_empty);
162 return;
163 }
164
165 /* Print CSV header first */
166 logging_csv_header();
167
168 /* Determine start position: oldest entry first */
169 if (log_count >= LOG_EEPROM_MAX)
170 pos = log_index; /* buffer full — oldest is at write_index */
171 else
172 pos = 0; /* buffer not full — oldest is at 0 */
173
174 for (i = 0; i < log_count; i++) {
175 base = ee_entry_addr(pos);
176
177 val = ee_read_float(base);
178 hour = eeprom_read_byte(base + 4);
179 min = eeprom_read_byte(base + 5);
180 sec = eeprom_read_byte(base + 6);
181 date = eeprom_read_byte(base + 7);
182 month = eeprom_read_byte(base + 8);
183 year = eeprom_read_byte(base + 9);
184
185 /* Unit is not stored in EEPROM — print placeholder */
186 print_entry(val, hour, min, sec, date, month, year, unit_none);
187
188 pos++;
189 if (pos >= LOG_EEPROM_MAX)
190 pos = 0;
191 }
192}
