73/* ─── Format float to string ─────────────────────────────────────── */
74
75static void float_to_str(float val, uint8_t decimals, char *buf, uint8_t len)
76{
77 /* Simple manual formatting to avoid heavy sprintf float */
78 int neg = 0;
79 if (val < 0.0f) {
80 neg = 1;
81 val = -val;
82 }
83
84 uint32_t integer = (uint32_t)val;
85 float frac = val - (float)integer;
86
87 /* Build fraction digits */
88 uint32_t frac_int = 0;
89 float mult = 1.0f;
90 for (uint8_t i = 0; i < decimals; i++)
91 mult *= 10.0f;
92 frac_int = (uint32_t)(frac * mult + 0.5f);
93
94 /* Handle rounding overflow */
95 if (decimals > 0) {
96 uint32_t frac_max = (uint32_t)mult;
97 if (frac_int >= frac_max) {
98 frac_int = 0;
99 integer++;
100 }
101 }
102
103 if (decimals > 0) {
104 snprintf_P(buf, len, neg ? PSTR("-%lu.%0*lu") : PSTR("%lu.%0*lu"),
105 (unsigned long)integer, decimals, (unsigned long)frac_int);
106 } else {
107 snprintf_P(buf, len, neg ? PSTR("-%lu") : PSTR("%lu"),
108 (unsigned long)integer);
109 }
110}

116void render_measurement(const meas_result_t *r)
117{
118 display_clear();
119
120 char buf[20];
121
122 /* ─── Status bar (y=0) ─────────────────────────────────────── */
123 get_mode_name(current_mode, buf, sizeof(buf));
124 display_str(0, 0, buf);
125
126 /* Range indicator */
127 if (range_locked) {
128 snprintf_P(buf, sizeof(buf), PSTR("R%u LCK"), r->range);
129 } else {
130 snprintf_P(buf, sizeof(buf), PSTR("R%u"), r->range);
131 }
132 display_str(50, 0, buf);
133
134 /* Function flags on the right */
135 uint8_t fx = 88;
136 if (current_func == FUNC_REL) { display_str_P(fx, 0, PSTR("REL")); fx += 24; }
137 if (current_func == FUNC_HOLD) { display_str_P(fx, 0, PSTR("HLD")); fx += 24; }
138 if (current_func == FUNC_MINMAX) { display_str_P(fx, 0, PSTR("M/M")); fx += 24; }
139 if (logging_enabled) { display_str_P(fx, 0, PSTR("LOG")); fx += 24; }
140 if (lpf_enabled) { display_str_P(fx, 0, PSTR("LPF")); }
141
142 /* Separator line */
143 display_hline(0, 9, 128);
144
145 /* ─── Primary reading (y=12, large font) ───────────────────── */
146 uint8_t dec = 3;
147 if (current_mode == M_FREQ || current_mode == M_TEMP)
148 dec = 2;
149 else if (current_mode == M_CONTIN || current_mode == M_DIODE)
150 dec = 3;
151 else if (current_mode == M_DBV || current_mode == M_DBM)
152 dec = 1;
153
154 float display_val = r->primary;
155
156 /* Overload display */
157 if (r->overload) {
158 display_str_large(4, 14, "OL");
159 } else {
160 float_to_str(display_val, dec, buf, sizeof(buf));
161 display_str_large(4, 14, buf);
162 }
163
164 /* Unit string to the right of the reading */
165 display_str(110, 20, r->unit);
166
167 /* Separator line */
168 display_hline(0, 39, 128);
169
170 /* ─── Secondary info (y=41) ────────────────────────────────── */
171 if (current_func == FUNC_MINMAX) {
172 /* Show Min/Max */
173 display_str_P(0, 41, PSTR("Mn:"));
174 float_to_str(min_val, 2, buf, sizeof(buf));
175 display_str(18, 41, buf);
176
177 display_str_P(68, 41, PSTR("Mx:"));
178 float_to_str(max_val, 2, buf, sizeof(buf));
179 display_str(86, 41, buf);
180 } else if (r->secondary != 0.0f) {
181 /* Show secondary value (e.g. AC component) */
182 float_to_str(r->secondary, 3, buf, sizeof(buf));
183 display_str(0, 41, buf);
184 }
185
186 /* ─── Analog bar graph (y=48-54) ───────────────────────────── */
187 /* Map primary to 0-100% based on expected range */
188 uint8_t percent = 0;
189 if (!r->overload && display_val >= 0.0f) {
190 /* Normalize to percentage (assume full scale around 5V / 100%) */
191 float fs;
192 switch (current_mode) {
193 case M_DCV: case M_ACV: case M_DCACV:
194 fs = 55.0f; break;
195 case M_MV_DC: case M_MV_AC:
196 fs = 5000.0f; break;
197 case M_TEMP:
198 fs = 110.0f; break;
199 case M_DUTY:
200 fs = 100.0f; break;
201 default:
202 fs = display_val * 2.0f; /* auto-scale */
203 if (fs < 1.0f) fs = 1.0f;
204 break;
205 }
206 float pct = display_val / fs * 100.0f;
207 if (pct > 100.0f) pct = 100.0f;
208 if (pct < 0.0f) pct = 0.0f;
209 percent = (uint8_t)pct;
210 }
211 display_bar(0, 48, 128, 6, percent);
212
213 /* ─── Footer (y=56) ────────────────────────────────────────── */
214 rtc_time_t t;
215 rtc_read(&t);
216 rtc_format_time(&t, buf);
217 display_str(0, 56, buf);
218
219 /* Log entry count if logging */
220 if (logging_enabled) {
221 display_str_P(80, 56, PSTR("LOG"));
222 }
223
224 /* AUTO or LOCK indicator */
225 if (range_locked)
226 display_str_P(104, 56, PSTR("LOCK"));
227 else
228 display_str_P(104, 56, PSTR("AUTO"));
229
230 display_flush();
231}
