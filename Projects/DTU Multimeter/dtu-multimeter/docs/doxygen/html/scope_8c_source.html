54/* Each entry: { ADCSRA prescaler bits, inter-sample delay in us }
55 * Index 0 = fastest (~77 ksps, prescaler /16, ~13 us/sample)
56 * Higher indices add inter-sample delays for slower timebases */

59 uint16_t delay_us; /* Additional delay between samples (us) */

112/* ═══════════════════════════════════════════════════════════════════
113 * Internal ADC Helpers (ATmega2560 ADC, channel 8 = A8)
114 * ═══════════════════════════════════════════════════════════════════ */
115
116static void internal_adc_init(void)
117{
118 /* AVCC reference, right-adjusted result */
119 ADMUX = (1 << REFS0);
120
121 /* Channel 8 → MUX[4:0] = 01000
122 * MUX[3:0] = 0000 in ADMUX (low nibble)
123 * MUX5 = 1 in ADCSRB for channels 8-15 */
124 ADMUX = (ADMUX & 0xE0) | (SCOPE_ADC_CH & 0x07);
125 ADCSRB = (SCOPE_ADC_CH >= 8) ? (1 << MUX5) : 0;
126
127 /* Enable ADC, prescaler /16 for fast conversion (~77 ksps)
128 * ADEN | ADPS2 = 0x84 + ADPS bits from timebase */
129 ADCSRA = (1 << ADEN) | 0x04; /* /16 default, updated per timebase */
130}
131
132static inline void internal_adc_set_prescaler(uint8_t adps)
133{
134 ADCSRA = (1 << ADEN) | (adps & 0x07);
135}
136
137static inline uint16_t internal_adc_read(void)
138{
139 ADCSRA |= (1 << ADSC); /* Start conversion */
140 while (ADCSRA & (1 << ADSC)) /* Wait for completion */
141 ;
142 return ADC; /* 10-bit result */
143}

218/* ═══════════════════════════════════════════════════════════════════
219 * Inter-sample delay (variable, from timebase table)
220 * ═══════════════════════════════════════════════════════════════════ */
221
222static void sample_delay(uint16_t us)
223{
224 /* _delay_us() needs a compile-time constant, so we loop in 10 us
225 * chunks for the variable portion. The granularity is acceptable
226 * for the slower timebases where this is used. */
227 while (us >= 10) {
228 _delay_us(10);
229 us -= 10;
230 }
231 /* Residual 0-9 us — close enough, the ADC conversion time
232 * already adds ~13 us of jitter at prescaler /16. */
233}

235/* ═══════════════════════════════════════════════════════════════════
236 * Capture Engine
237 * ═══════════════════════════════════════════════════════════════════ */
238
239void scope_capture(void)
240{
241 timebase_cfg_t tb_cfg;
242 tb_read(scope_timebase, &tb_cfg);
243
244 /* Apply ADC prescaler for this timebase */
245 if (scope_use_internal_adc)
246 internal_adc_set_prescaler(tb_cfg.adps);
247
248 /* ── Step 1: AC coupling DC offset removal ────────────────────── */
249
250 if (scope_ac_coupling) {
251 uint32_t sum = 0;
252 for (uint8_t i = 0; i < 64; i++)
253 sum += scope_read_sample();
254 dc_offset = (uint16_t)(sum >> 6);
255 } else {
256 dc_offset = 0;
257 }
258
259 /* ── Step 2: Fill pre-trigger buffer (circular) ───────────────── */
260
261 uint16_t pre_idx = 0;
262 for (uint16_t i = 0; i < SCOPE_PRETRIG; i++) {
263 scope_buf[i] = scope_read_sample();
264 if (tb_cfg.delay_us)
265 sample_delay(tb_cfg.delay_us);
266 }
267 pre_idx = SCOPE_PRETRIG; /* Next write position in pre-trig area */
268
269 /* ── Step 3: Wait for trigger ─────────────────────────────────── */
270
271 uint8_t triggered = 0;
272 uint16_t trig_pos = 0; /* Index in scope_buf where trigger found */
273 uint16_t prev_sample = scope_buf[SCOPE_PRETRIG - 1];
274 uint32_t trig_timeout = millis() + 500; /* 500 ms auto-trigger timeout */
275
276 if (scope_trig_mode == TRIG_NONE) {
277 /* Free-run: no triggering, start post-trigger immediately */
278 triggered = 1;
279 trig_pos = SCOPE_PRETRIG;
280 } else {
281 /* Scan for edge crossing trigger level in incoming samples */
282 while (!triggered) {
283 uint16_t s = scope_read_sample();
284 if (tb_cfg.delay_us)
285 sample_delay(tb_cfg.delay_us);
286
287 /* Overwrite oldest pre-trigger sample (circular) */
288 if (pre_idx >= SCOPE_PRETRIG)
289 pre_idx = 0;
290 scope_buf[pre_idx] = s;
291 pre_idx++;
292
293 /* Edge detection */
294 if (scope_trig_mode == TRIG_RISING || scope_trig_mode == TRIG_AUTO ||
295 scope_trig_mode == TRIG_SINGLE) {
296 /* Rising edge: previous below level, current at or above */
297 if (prev_sample < scope_trig_level && s >= scope_trig_level) {
298 triggered = 1;
299 trig_pos = SCOPE_PRETRIG;
300 }
301 }
302 if (!triggered &&
303 (scope_trig_mode == TRIG_FALLING)) {
304 /* Falling edge: previous above level, current at or below */
305 if (prev_sample > scope_trig_level && s <= scope_trig_level) {
306 triggered = 1;
307 trig_pos = SCOPE_PRETRIG;
308 }
309 }
310
311 prev_sample = s;
312
313 /* Auto mode: capture anyway on timeout */
314 if (!triggered && (scope_trig_mode == TRIG_AUTO || scope_trig_mode == TRIG_SINGLE)) {
315 if (millis() > trig_timeout) {
316 triggered = 1;
317 trig_pos = SCOPE_PRETRIG;
318 }
319 }
320 }
321
322 /* Unroll circular pre-trigger buffer so sample 0 is oldest.
323 * pre_idx points to the next write slot = oldest sample. */
324 if (pre_idx < SCOPE_PRETRIG && pre_idx != 0) {
325 uint16_t tmp[SCOPE_PRETRIG];
326 memcpy(tmp, scope_buf, SCOPE_PRETRIG * sizeof(uint16_t));
327 uint16_t tail = SCOPE_PRETRIG - pre_idx;
328 memcpy(scope_buf, &tmp[pre_idx], tail * sizeof(uint16_t));
329 memcpy(&scope_buf[tail], tmp, pre_idx * sizeof(uint16_t));
330 }
331 }
332
333 /* ── Step 4: Fill post-trigger samples ────────────────────────── */
334
335 uint16_t post_count = SCOPE_BUF_SIZE - SCOPE_PRETRIG;
336 for (uint16_t i = 0; i < post_count; i++) {
337 scope_buf[SCOPE_PRETRIG + i] = scope_read_sample();
338 if (tb_cfg.delay_us)
339 sample_delay(tb_cfg.delay_us);
340 }
341
342 /* ── Step 5: Apply AC coupling offset removal ─────────────────── */
343
344 if (scope_ac_coupling && dc_offset > 0) {
345 for (uint16_t i = 0; i < SCOPE_BUF_SIZE; i++) {
346 int16_t val = (int16_t)scope_buf[i] - (int16_t)dc_offset + 512;
347 if (val < 0) val = 0;
348 if (val > 1023) val = 1023;
349 scope_buf[i] = (uint16_t)val;
350 }
351 }
352}

365/* ═══════════════════════════════════════════════════════════════════
366 * Auto-Measurements (Vpp, Vavg, Vrms, Frequency, Duty Cycle)
367 * ═══════════════════════════════════════════════════════════════════ */
368
369void scope_auto_measure(scope_meas_t *m)
370{
371 uint16_t min_val = 1023;
372 uint16_t max_val = 0;
373 uint32_t sum = 0;
374 double sum_sq = 0.0;
375
376 /* ── Pass 1: min, max, sum, sum-of-squares ────────────────────── */
377
378 for (uint16_t i = 0; i < SCOPE_BUF_SIZE; i++) {
379 uint16_t s = scope_buf[i];
380 if (s < min_val) min_val = s;
381 if (s > max_val) max_val = s;
382 sum += s;
383 sum_sq += (double)s * (double)s;
384 }
385
386 float scale = V_REF / 1023.0f;
387
388 /* Vpp */
389 m->vpp = (float)(max_val - min_val) * scale;
390
391 /* Vavg */
392 float avg_raw = (float)sum / (float)SCOPE_BUF_SIZE;
393 m->vavg = avg_raw * scale;
394
395 /* Vrms */
396 float mean_sq = (float)(sum_sq / (double)SCOPE_BUF_SIZE);
397 m->vrms = sqrtf(mean_sq) * scale;
398
399 /* ── Pass 2: zero-crossing frequency and duty cycle ───────────── */
400
401 uint16_t midpoint = (min_val + max_val) / 2;
402 uint16_t crossings = 0;
403 uint16_t above_count = 0;
404
405 for (uint16_t i = 1; i < SCOPE_BUF_SIZE; i++) {
406 /* Count rising zero crossings around midpoint */
407 if (scope_buf[i - 1] < midpoint && scope_buf[i] >= midpoint)
408 crossings++;
409 if (scope_buf[i] >= midpoint)
410 above_count++;
411 }
412 /* Check first sample too for duty */
413 if (scope_buf[0] >= midpoint)
414 above_count++;
415
416 /* Sample period from timebase table */
417 float t_sample = pgm_read_float_near(&tb_period_us[scope_timebase]);
418 float total_time_s = t_sample * (float)SCOPE_BUF_SIZE / 1e6f;
419
420 /* Frequency: rising crossings / total time */
421 if (crossings >= 2)
422 m->freq = (float)crossings / total_time_s;
423 else
424 m->freq = 0.0f;
425
426 /* Duty cycle: percentage of samples above midpoint */
427 m->duty = (float)above_count / (float)SCOPE_BUF_SIZE * 100.0f;
428}

430/* ═══════════════════════════════════════════════════════════════════
431 * OLED Rendering (128x64 SSD1306 framebuffer)
432 * ═══════════════════════════════════════════════════════════════════ */

434/* Waveform area: x=0..127, y=10..49 (40 pixels height)
435 * Top 10 rows: status text
436 * Bottom 14 rows: measurement readout */
437
438#define WAVE_Y_MIN 10
439#define WAVE_Y_MAX 49
440#define WAVE_HEIGHT (WAVE_Y_MAX - WAVE_Y_MIN)
441#define OLED_COLS 128
442
443static inline void fb_set_pixel(uint8_t *fb, uint8_t x, uint8_t y)
444{
445 if (x >= OLED_WIDTH || y >= OLED_HEIGHT)
446 return;
447 fb[x + (y / 8) * OLED_WIDTH] |= (1 << (y & 7));
448}
449
450static void fb_draw_char(uint8_t *fb, uint8_t x, uint8_t y, char c)
451{
452 /* Delegate to display module — we just set pixels for waveform */
453 (void)fb; (void)x; (void)y; (void)c;
454 /* Text rendering uses display_char() on the shared framebuffer */
455}
456
457void scope_render_oled(uint8_t *fb)
458{
459 /* ── Find buffer min/max for scaling ──────────────────────────── */
460
461 uint16_t buf_min = 1023, buf_max = 0;
462 for (uint16_t i = 0; i < SCOPE_BUF_SIZE; i++) {
463 if (scope_buf[i] < buf_min) buf_min = scope_buf[i];
464 if (scope_buf[i] > buf_max) buf_max = scope_buf[i];
465 }
466 uint16_t range = buf_max - buf_min;
467 if (range == 0) range = 1; /* Avoid division by zero */
468
469 /* ── Draw grid dots (every 16 pixels horizontally, 10 vertically) */
470
471 for (uint8_t gx = 0; gx < OLED_COLS; gx += 16) {
472 for (uint8_t gy = WAVE_Y_MIN; gy <= WAVE_Y_MAX; gy += 10)
473 fb_set_pixel(fb, gx, gy);
474 }
475
476 /* ── Draw trigger level dashed line ───────────────────────────── */
477
478 uint8_t trig_y;
479 {
480 int16_t tl = (int16_t)scope_trig_level - (int16_t)buf_min;
481 if (tl < 0) tl = 0;
482 if (tl > (int16_t)range) tl = (int16_t)range;
483 trig_y = WAVE_Y_MAX - (uint8_t)((uint32_t)tl * WAVE_HEIGHT / range);
484 }
485 for (uint8_t x = 0; x < OLED_COLS; x += 4)
486 fb_set_pixel(fb, x, trig_y);
487
488 /* ── Draw waveform: 128 columns from 1000 samples (decimate) ─── */
489
490 uint8_t prev_y = 0;
491 for (uint8_t x = 0; x < OLED_COLS; x++) {
492 /* Map column x to buffer index */
493 uint16_t buf_idx = (uint16_t)x * SCOPE_BUF_SIZE / OLED_COLS;
494 if (buf_idx >= SCOPE_BUF_SIZE) buf_idx = SCOPE_BUF_SIZE - 1;
495
496 /* Scale sample to waveform Y range (inverted: higher value = lower Y) */
497 int16_t s = (int16_t)scope_buf[buf_idx] - (int16_t)buf_min;
498 uint8_t y = WAVE_Y_MAX - (uint8_t)((uint32_t)s * WAVE_HEIGHT / range);
499
500 /* Clamp */
501 if (y < WAVE_Y_MIN) y = WAVE_Y_MIN;
502 if (y > WAVE_Y_MAX) y = WAVE_Y_MAX;
503
504 fb_set_pixel(fb, x, y);
505
506 /* Draw vertical line segment between consecutive points for
507 * continuous waveform appearance */
508 if (x > 0) {
509 uint8_t y0 = (prev_y < y) ? prev_y : y;
510 uint8_t y1 = (prev_y < y) ? y : prev_y;
511 for (uint8_t yy = y0 + 1; yy < y1; yy++)
512 fb_set_pixel(fb, x, yy);
513 }
514 prev_y = y;
515 }
516
517 /* ── Top status bar: trigger mode, timebase, trigger level ────── */
518
519 {
520 /* Trigger mode name */
521 const char *tname;
522 memcpy_P(&tname, &trig_names[scope_trig_mode], sizeof(const char *));
523 display_str_P(0, 0, tname);
524
525 /* Timebase index */
526 display_str_P(30, 0, str_tb_label);
527 display_char(48, 0, '0' + scope_timebase);
528
529 /* Trigger level */
530 display_str_P(60, 0, str_tl_label);
531 /* 4-digit trigger level number */
532 uint16_t tl = scope_trig_level;
533 display_char(78, 0, '0' + (tl / 1000) % 10);
534 display_char(84, 0, '0' + (tl / 100) % 10);
535 display_char(90, 0, '0' + (tl / 10) % 10);
536 display_char(96, 0, '0' + tl % 10);
537 }
538
539 /* ── Bottom: Vpp and frequency ────────────────────────────────── */
540
541 {
542 scope_meas_t m;
543 scope_auto_measure(&m);
544
545 display_str_P(0, 56, str_vpp_label);
546 /* Simple Vpp display: X.XX V */
547 uint16_t vpp_mv = (uint16_t)(m.vpp * 1000.0f);
548 display_char(24, 56, '0' + (vpp_mv / 1000) % 10);
549 display_char(30, 56, '.');
550 display_char(36, 56, '0' + (vpp_mv / 100) % 10);
551 display_char(42, 56, '0' + (vpp_mv / 10) % 10);
552 display_str_P(48, 56, str_v_unit);
553
554 display_str_P(64, 56, str_freq_label);
555 if (m.freq >= 1000.0f) {
556 uint16_t f_10 = (uint16_t)(m.freq / 100.0f); /* freq in 0.1 kHz */
557 display_char(76, 56, '0' + (f_10 / 100) % 10);
558 display_char(82, 56, '0' + (f_10 / 10) % 10);
559 display_char(88, 56, '.');
560 display_char(94, 56, '0' + f_10 % 10);
561 display_str_P(100, 56, str_khz_unit);
562 } else {
563 uint16_t f_hz = (uint16_t)m.freq;
564 display_char(76, 56, '0' + (f_hz / 10000) % 10);
565 display_char(82, 56, '0' + (f_hz / 1000) % 10);
566 display_char(88, 56, '0' + (f_hz / 100) % 10);
567 display_char(94, 56, '0' + (f_hz / 10) % 10);
568 display_char(100, 56, '0' + f_hz % 10);
569 display_str_P(106, 56, str_hz_unit);
570 }
571 }

574/* ═══════════════════════════════════════════════════════════════════
575 * ASCII Waveform (80x24) via UART
576 * ═══════════════════════════════════════════════════════════════════ */
577
578#define ASCII_COLS 80
579#define ASCII_ROWS 24
580
581void scope_print_ascii(void)
582{
583 /* Find buffer min/max */
584 uint16_t buf_min = 1023, buf_max = 0;
585 for (uint16_t i = 0; i < SCOPE_BUF_SIZE; i++) {
586 if (scope_buf[i] < buf_min) buf_min = scope_buf[i];
587 if (scope_buf[i] > buf_max) buf_max = scope_buf[i];
588 }
589 uint16_t range = buf_max - buf_min;
590 if (range == 0) range = 1;
591
592 /* Build row-mapped column array (which row each column maps to) */
593 uint8_t col_row[ASCII_COLS];
594 for (uint8_t c = 0; c < ASCII_COLS; c++) {
595 uint16_t idx = (uint16_t)c * SCOPE_BUF_SIZE / ASCII_COLS;
596 if (idx >= SCOPE_BUF_SIZE) idx = SCOPE_BUF_SIZE - 1;
597 int16_t s = (int16_t)scope_buf[idx] - (int16_t)buf_min;
598 /* Row 0 = top = max, row (ROWS-1) = bottom = min */
599 col_row[c] = (ASCII_ROWS - 1) - (uint8_t)((uint32_t)s * (ASCII_ROWS - 1) / range);
600 }
601
602 /* Trigger level row */
603 int16_t tl_s = (int16_t)scope_trig_level - (int16_t)buf_min;
604 if (tl_s < 0) tl_s = 0;
605 if (tl_s > (int16_t)range) tl_s = (int16_t)range;
606 uint8_t trig_row = (ASCII_ROWS - 1) - (uint8_t)((uint32_t)tl_s * (ASCII_ROWS - 1) / range);
607
608 /* Print top border */
609 uart_puts_P(str_ascii_top);
610 for (uint8_t c = 0; c < ASCII_COLS; c++)
611 uart_puts_P(str_ascii_dash);
612 uart_puts_P(str_ascii_top);
613 uart_puts_P(str_newline);
614
615 /* Print each row */
616 for (uint8_t r = 0; r < ASCII_ROWS; r++) {
617 uart_puts_P(str_ascii_bar);
618 for (uint8_t c = 0; c < ASCII_COLS; c++) {
619 if (col_row[c] == r)
620 uart_putc('*');
621 else if (r == trig_row && (c & 3) == 0)
622 uart_putc('-');
623 else if (c == 0 || c == ASCII_COLS / 2)
624 uart_putc(':');
625 else if (r == ASCII_ROWS / 2 && (c & 7) == 0)
626 uart_putc('.');
627 else
628 uart_putc(' ');
629 }
630 uart_puts_P(str_ascii_bar);
631 uart_puts_P(str_newline);
632 }
633
634 /* Print bottom border */
635 uart_puts_P(str_ascii_top);
636 for (uint8_t c = 0; c < ASCII_COLS; c++)
637 uart_puts_P(str_ascii_dash);
638 uart_puts_P(str_ascii_top);
639 uart_puts_P(str_newline);
640
641 /* Print measurements below */
642 scope_meas_t m;
643 scope_auto_measure(&m);
644
645 uart_puts_P(str_vpp_label);
646 uart_print_float(m.vpp, 3);
647 uart_puts_P(str_v_unit);
648 uart_putc(' ');
649 uart_puts_P(str_freq_label);
650 uart_print_float(m.freq, 1);
651 uart_puts_P(str_hz_unit);
652 uart_puts_P(str_newline);
653}
